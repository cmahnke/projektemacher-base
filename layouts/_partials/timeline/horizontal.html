{{- $entries := .entries -}} {{- /* Front matter param holding the year, sites can override (kinderbuecher: issueYear). Year values may be plain numbers or range strings like '1922-1924', '1929-31' or '1921–1923' (en dash), ranges become timeline items with a start/end date. */ -}} {{- $yearKey := $.yearKey | default "year" -}}
{{- $items := slice -}} {{- range $entries -}} {{- $yearRaw := index .Params $yearKey -}} {{- if and (ne $yearRaw nil) (findRE "^\\s*\\d{4}" (printf "%v" $yearRaw)) -}} {{- $yearStr := printf "%v" $yearRaw -}} {{- $year := int (replaceRE "\\s" "" (index (findRE "^\\s*\\d{4}" $yearStr) 0)) -}} {{- $yearEnd := $year -}} {{- $rangeMatch := findRE "[-–—]\\s*\\d{2,4}\\s*$" $yearStr -}} {{- if gt (len $rangeMatch) 0 -}} {{- $suffix := replaceRE "[^\\d]" "" (index $rangeMatch 0) -}} {{- if ge (len $suffix) 3 -}} {{- $yearEnd = int $suffix -}} {{- else -}} {{- $expanded := add (mul (div $year 100) 100) (int $suffix) -}} {{- if lt $expanded $year -}} {{- $expanded = add $expanded 100 -}} {{- end -}} {{- if gt $expanded $year -}} {{- $yearEnd = $expanded -}} {{- end -}} {{- end -}} {{- end -}} {{- $items = $items | append (dict "year" $year "yearEnd" $yearEnd "entry" .) -}} {{- end -}} {{- end -}}
