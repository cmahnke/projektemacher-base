{{- /* Expands the strict front-matter structure shorthand (`.Params.iiif.structure`) into IIIF Presentation `structures`. Called from `iiif/3/common.json` and `iiif/existing-manifest.html`. Contract (per entry): label: {de: ["Titel"]} required, IIIF label object, coerced to strings page: page012 reference to one of the page's `resources`, resolved by the resource's `name`, the `params.iiif` directory, or the `src` basename (without extension) - the same chain `iiif/3/manifest.json` uses for canvas names end: page015 optional span end (inclusive), expands to the canvas list from `page` to `end` items: optional nested entries with the same shape; an entry without `page` acts as a grouping node The output follows the IIIF Cookbook recipe 0024 "Table of Contents for Book Chapters" (https://iiif.io/api/cookbook/recipe/0024-book-4-toc/): a single root range labeled via the `iiifTableOfContents` i18n key (no `behavior`) whose `items` hold the top-level entries. Ranges with sub-ranges contain ONLY those sub-ranges; leaf ranges contain ALL canvases of their span. Unlike earlier emissions there are no first/last canvas markers: that pattern is not part of the recipe and Tify <= 0.36.2 renders the extra canvas references as unexpected table-of-contents rows between the sub-chapters. Accepted viewer consequences on Tify <= 0.36.2 (its span heuristic derives firstPage/lastPage from the first and last `items` entry without descending into sub-ranges, store.js TODO "dig deeper"): the previous/next-section pagination buttons stay hidden and the Info view shows no current chapter for range-only branches. The root is always emitted because a top-level forest of range-only containers would be merged into the first container by Tify's span-containment heuristic and dropped from the top level. Range and Canvas identifiers are generated at build time (hierarchical, depth first: range/toc, range/0, range/0-1, ...) using the same deployment prefix computation as `iiif/3/manifest.json`, so they always match the manifest's own canvas identifiers in every environment. Hand-written `id`s or legacy IIIF Range entries abort the build. Only page bundles are expanded; sections keep their raw value (leaflet mode computes its own ranges and warns, see `manifest-leaflet.json`). Uses a single top-level `return`: Hugo < 0.166 supports only one `return` statement with a value per partial. */ -}} {{- $page := .page -}} {{- $entries := .entries -}} {{- $urlBase := .urlBase | default "" -}} {{- $canvasNames := .canvasNames | default slice -}} {{- $refIndex := .refIndex | default dict -}} {{- $prefix := .prefix | default "" -}} {{- $isRoot := true -}} {{- if isset . "root" -}} {{- $isRoot = .root -}} {{- end -}} {{- $out := slice -}} {{- if ne $page.Kind "page" -}} {{- with $entries -}} {{- warnf "[iiif/3/structures.html] iiif.structure is only supported on pages, keeping the raw value for %s" $page.RelPermalink -}} {{- end -}} {{- $out = $entries -}} {{- else -}} {{- if eq (len $canvasNames) 0 -}} {{- /* Deployment prefix, identical to layouts/partials/iiif/3/manifest.json */ -}} {{- $defaultContentPage := $page -}} {{- if hugo.IsMultilingual -}} {{- $defaultContentPage = index (where $page.AllTranslations ".Lang" "eq" hugo.Sites.Default.Language.Lang) 0 -}} {{- end -}} {{- $localPath := printf "%s/" (strings.Substr (path.Dir $defaultContentPage.RelPermalink) 1) -}} {{- if or hugo.IsServer (not (isset site.Params "productioniiifurlprefix")) -}} {{- $urlBase = printf "%s%s" hugo.Sites.Default.BaseURL $localPath -}} {{- else -}} {{- $urlBase = printf "%s%s" site.Params.productionIIIFURLPrefix $localPath -}} {{- end -}} {{- /* Reference lookup: ordered canvas names + ref -> resource index */ -}} {{- range $i, $res := $page.Params.resources -}} {{- $src := printf "%v" ($res.src | default "") -}} {{- $name := printf "%v" ($res.name | default "") -}} {{- $iiifDir := "" -}} {{- with $res.params -}} {{- with index . "iiif" -}} {{- $iiifDir = path.BaseName (path.Dir (printf "%v" .)) -}} {{- end -}} {{- end -}} {{- $isMedia := false -}} {{- with $page.Resources.Get $src -}} {{- if in (slice "image" "audio" "model") (index (split (string .MediaType) "/") 0) -}} {{- $isMedia = true -}} {{- end -}} {{- end -}} {{- if $isMedia -}} {{- $canvas := $name -}} {{- if eq $canvas "" -}} {{- if ne $iiifDir "" -}} {{- $canvas = $iiifDir -}} {{- else -}} {{- $canvas = path.BaseName $src -}} {{- end -}} {{- end -}} {{- $canvasNames = $canvasNames | append $canvas -}} {{- range slice (path.BaseName $src) $name $iiifDir -}} {{- if ne . "" -}} {{- if not (isset $refIndex .) -}} {{- $refIndex = merge $refIndex (dict . $i) -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $canvasNames = $canvasNames | append "" -}} {{- end -}} {{- end -}} {{- end -}} {{- range $i, $entry := $entries -}} {{- if not (reflect.IsMap $entry) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d is not an object (%s)" $i $page.RelPermalink -}} {{- end -}} {{- with index $entry "id" -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has a hand-written id '%v', identifiers are generated - use the label/page/items shorthand (%s)" $i . $page.RelPermalink -}} {{- end -}} {{- $label := index $entry "label" -}} {{- if or (not $label) (not (reflect.IsMap $label)) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has no label object (%s)" $i $page.RelPermalink -}} {{- end -}} {{- $label = partial "functions/iiif/coerce-language-map.html" $label -}} {{- /* Own page validation: containers validate their declared page/end but only leaves emit canvases from it; containers hold only their sub-ranges, like cookbook recipe 0024 */ -}} {{- $ownStart := -1 -}} {{- $ownEnd := -1 -}} {{- $ref := printf "%v" ($entry.page | default "") -}} {{- if ne $ref "" -}} {{- if not (isset $refIndex $ref) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d references unknown page '%v' (%s)" $i $ref $page.RelPermalink -}} {{- end -}} {{- $ownStart = index $refIndex $ref -}} {{- $ownEnd = $ownStart -}} {{- $endRef := printf "%v" ($entry.end | default "") -}} {{- if ne $endRef "" -}} {{- if not (isset $refIndex $endRef) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d references unknown end page '%v' (%s)" $i $endRef $page.RelPermalink -}} {{- end -}} {{- $ownEnd = index $refIndex $endRef -}} {{- if lt $ownEnd $ownStart -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has end page '%v' before start page '%v' (%s)" $i $endRef $ref $page.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- $items := slice -}} {{- $hasChildren := false -}} {{- with index $entry "items" -}} {{- $hasChildren = true -}} {{- end -}} {{- if not $hasChildren -}} {{- /* Leaf: all canvases of the own span */ -}} {{- if ge $ownStart 0 -}} {{- range $j := seq $ownStart $ownEnd -}} {{- $canvas := index $canvasNames $j -}} {{- if ne $canvas "" -}} {{- $items = $items | append (dict "id" (printf "%scanvas/%s" $urlBase $canvas) "type" "Canvas") -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Container: ranges only. Warn when the declared own span reaches beyond the children's coverage - those canvases would be unreachable through this range */ -}} {{- $childSpan := partial "functions/iiif/structure-span.html" (dict "entry" (dict "items" (index $entry "items")) "refIndex" $refIndex "fallback" -1) -}} {{- if ge $ownStart 0 -}} {{- $childStart := index $childSpan "start" -}} {{- $childEnd := index $childSpan "end" -}} {{- if or (lt $childStart 0) (gt $childStart $ownStart) (lt $childEnd $ownEnd) -}} {{- warnf "[iiif/3/structures.html] structure entry #%d declares page '%v' beyond its children's coverage; uncovered pages are unreachable through the range (%s)" $i $ref $page.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- with index $entry "items" -}} {{- range $sub := (partial "iiif/3/structures.html" (dict "page" $page "entries" . "urlBase" $urlBase "canvasNames" $canvasNames "refIndex" $refIndex "prefix" (printf "%s%d-" $prefix $i) "root" false)) -}} {{- $items = $items | append $sub -}} {{- end -}} {{- end -}} {{- if eq (len $items) 0 -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has neither a resolvable page nor nested items (%s)" $i $page.RelPermalink -}} {{- end -}} {{- $out = $out | append (dict "id" (printf "%srange/%s%d" $urlBase $prefix $i) "type" "Range" "label" $label "items" $items) -}} {{- end -}} {{- /* Cookbook 0024 root: one wrapper range holding the structure tree, localized label, no behavior. Recursions (root=false) skip this. */ -}} {{- if and $isRoot (gt (len $out) 0) -}} {{- $wrapperLabel := i18n "iiifTableOfContents" -}} {{- if not $wrapperLabel -}} {{- $wrapperLabel = "Table of Contents" -}} {{- end -}} {{- $out = slice (dict "id" (printf "%srange/toc" $urlBase) "type" "Range" "label" (dict ($page.Language.Lang | printf "%v") (slice $wrapperLabel)) "items" $out) -}} {{- end -}} {{- end -}} {{- return $out -}}
