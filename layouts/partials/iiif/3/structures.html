{{- /* Expands the strict front-matter structure shorthand (`.Params.iiif.structure`) into IIIF Presentation `structures`. Called from `iiif/3/common.json`. Contract (per entry): label: {de: ["Titel"]} required, IIIF label object, coerced to strings page: page012 reference to one of the page's `resources`, resolved by the resource's `name`, the `params.iiif` directory, or the `src` basename (without extension) - the same chain `iiif/3/manifest.json` uses for canvas names end: page015 optional span end (inclusive), expands to the canvas list from `page` to `end` items: optional nested entries with the same shape; an entry without `page` acts as a grouping node Entries with nested `items` always span the full canvas range covered by themselves and their descendants: the first item is a Canvas reference and the last item is a Canvas reference wrapping the inline child ranges (grouping nodes without `page` get both). This keeps the hierarchy spec-valid while remaining compatible with Tify's table of contents heuristic, which derives a range's page span from the first and last `items` entry and requires both to be canvases. Range and Canvas identifiers are generated at build time (hierarchical, depth first: range/0, range/0-1, ...) using the same deployment prefix computation as `iiif/3/manifest.json`, so they always match the manifest's own canvas identifiers in every environment. Hand-written `id`s or legacy IIIF Range entries abort the build. Only page bundles are expanded; sections keep their raw value (leaflet mode computes its own ranges and warns, see `manifest-leaflet.json`). Uses a single top-level `return`: Hugo < 0.166 supports only one `return` statement with a value per partial. */ -}} {{- $page := .page -}} {{- $entries := .entries -}} {{- $urlBase := .urlBase | default "" -}} {{- $canvasNames := .canvasNames | default slice -}} {{- $refIndex := .refIndex | default dict -}} {{- $prefix := .prefix | default "" -}} {{- $out := slice -}} {{- if ne $page.Kind "page" -}} {{- with $entries -}} {{- warnf "[iiif/3/structures.html] iiif.structure is only supported on pages, keeping the raw value for %s" $page.RelPermalink -}} {{- end -}} {{- $out = $entries -}} {{- else -}} {{- if eq (len $canvasNames) 0 -}} {{- /* Deployment prefix, identical to layouts/partials/iiif/3/manifest.json */ -}} {{- $defaultContentPage := $page -}} {{- if hugo.IsMultilingual -}} {{- $defaultContentPage = index (where $page.AllTranslations ".Lang" "eq" hugo.Sites.Default.Language.Lang) 0 -}} {{- end -}} {{- $localPath := printf "%s/" (substr (path.Dir $defaultContentPage.RelPermalink) 1) -}} {{- if or hugo.IsServer (not (isset $page.Site.Params "productioniiifurlprefix")) -}} {{- $urlBase = printf "%s%s" hugo.Sites.Default.BaseURL $localPath -}} {{- else -}} {{- $urlBase = printf "%s%s" $page.Site.Params.productionIIIFURLPrefix $localPath -}} {{- end -}} {{- /* Reference lookup: ordered canvas names + ref -> resource index */ -}} {{- range $i, $res := $page.Params.resources -}} {{- $src := printf "%v" ($res.src | default "") -}} {{- $name := printf "%v" ($res.name | default "") -}} {{- $iiifDir := "" -}} {{- with $res.params -}} {{- with index . "" -}} {{- if not (isset $refIndex .) -}} {{- $refIndex = merge $refIndex (dict . $i) -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- $canvasNames = $canvasNames | append "" -}} {{- end -}} {{- end -}} {{- end -}} {{- range $i, $entry := $entries -}} {{- if not (reflect.IsMap $entry) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d is not an object (%s)" $i $page.RelPermalink -}} {{- end -}} {{- with index $entry "id" -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has a hand-written id '%v', identifiers are generated - use the label/page/items shorthand (%s)" $i . $page.RelPermalink -}} {{- end -}} {{- with index $entry "type" -}} {{- errorf "[iiif/3/structures.html] structure entry #%d uses the legacy 'type: %v' format - use the label/page/items shorthand (%s)" $i . $page.RelPermalink -}} {{- end -}} {{- $label := index $entry "label" -}} {{- if or (not $label) (not (reflect.IsMap $label)) -}} {{- errorf "[iiif/3/structures.html] structure entry #%d has no label object (%s)" $i $page.RelPermalink -}} {{- end -}} {{- $label = partial "functions/iiif/coerce-language-map.html" $label -}} {{- $items := slice -}} {{- $ref := printf "%v" ($entry.page | $refIndex $ref -}} {{- $endIdx := $startIdx -}} {{- $ownEndIdx = $startIdx -}} {{- $endRef := printf "%v" ($entry.end | default "") -}} {{- if ne $endRef "" -}} {{- if not (isset $refIndex $endRef) $sub := (partial "iiif/3/structures.html" (dict "page" $page "entries" .
