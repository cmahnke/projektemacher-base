{{/* Hugo Shortcode for browsing IIIF Collections. The browser is selected with the 'browser' parameter, defaulting to 'kasane' (overridable site-wide via '[params.iiif] collectionBrowser'): * kasane - the Kasane viewer (@iiif-kasane, web component) with its collection browser in the structure side panel: items can be opened directly, nested collections are navigated, ordering can be switched between 'as listed' and 'by date' (with a timeline view). * mango - a tree navigator (@mango-iiif/collection-navigator) with the Mango viewer (@mango-iiif/iiif-viewer), loaded in-place without reloading the page. The Mango bundle is built with esbuild code splitting, keeping the lazy-loaded renderers (AV, PDF, 3D) out of the entry chunk. Nested collections are expanded lazily, only expanded branches are loaded. Parameters: * collectionUrl - The url to the collection. Relative urls are resolved against the page permalink. Defaults to the collection generated for the site's home page. * browser - 'kasane' (default) or 'mango' * height - Optional CSS height of the browser area, defaults to '80vh' (mango); the kasane browser keeps an A4 page aspect ratio (1 / 1.414) on desktop and a fixed height ('24rem') on mobile unless a height is given * locale - Viewer locale, 'ja' or 'en' (kasane only, default 'en') * class - Additional CSS classes for the wrapper (kasane only) * options - JSON string with viewer options (kasane only, see https://github.com/nakamura196/iiif-kasane README) */}} {{- $collectionUrl := .Get "collectionUrl" -}} {{- $urlBase := $.Page.Permalink | replaceRE "\\.html$" "/" -}} {{- if $collectionUrl -}} {{- if not (or (hasPrefix $collectionUrl "http://") (hasPrefix $collectionUrl "https://")) -}} {{- $collectionUrl = printf "%s%s" $urlBase $collectionUrl -}} {{- end -}} {{- else -}} {{- with site.Home.OutputFormats.Get "iiif-collection" -}} {{- $collectionUrl = .Permalink -}} {{- else -}} {{- errorf "The %q shortcode requires a 'collectionUrl' or the site home to generate an 'iiif-collection' output" .Name -}} {{- end -}} {{- end -}} {{- $browser := "kasane" -}} {{- with site.Params.iiif -}} {{- with .collectionBrowser -}} {{- $browser = . -}} {{- end -}} {{- end -}} {{- with .Get "browser" -}} {{- $browser = . -}} {{- end -}} {{- if not (in (slice "kasane" "mango") $browser) -}} {{- errorf "The %q shortcode parameter 'browser' must be one of 'kasane' or 'mango', got %q" .Name $browser -}} {{- end -}} {{- $height := "" -}} {{- with .Get "height" -}} {{- $height = . -}} {{- end -}} {{- if eq $browser "kasane" -}} {{- partial "shortcodes/iiif/kasane-viewer.html" (dict "content" $collectionUrl "locale" (.Get "locale") "class" (.Get "class") "options" (.Get "options") "height" $height) -}} {{- else -}} {{- $heightAttr := "" -}} {{- with $height -}} {{- $heightAttr = printf " style=\"--collection-browser-height:%s\"" . -}} {{- end -}} {{- $id := printf "_iiif-collection-%s" (md5 (printf "%s-%d" $collectionUrl .Ordinal)) -}} {{- $jsEntry := resources.Get "ts/iiif/collection.ts" -}} {{- $jsOpts := dict "format" "esm" "splitting" true "minify" true "target" "es2022" "targetPath" "js/mango-viewer.js" -}} {{- $js := $jsEntry | js.Build $jsOpts | fingerprint -}}
{{- end -}} {{- define "_partials/shortcodes/iiif/collection-metadata.html" -}} {{- (dict "js" (slice) "scss" (slice "scss/iiif/collection-mango.scss" "scss/iiif/kasane.scss")) | jsonify -}} {{- end -}}
