{{/* Renders the theme's i18n strings in a single table with one column per language, each with a client side search in its table header. The data source is the theme's i18n directory mounted as data (see hugo.toml: module.mounts '../i18n' -> 'data/i18n'), so the page updates automatically when translations are added or changed. Recognized file names (Hugo data keys, extension stripped): - '.toml' — merged language file - '..toml' — i18n fragment (merged into '.toml' by scripts/init/i18n.sh at build time) */}} {{- $data := .Site.Data.i18n -}} {{- if not $data -}} {{- errorf "[shortcodes/i18n-strings.html] No i18n data found — is the module mount 'i18n' -> 'data/i18n' configured?" -}} {{- end -}} {{- /* Collect the languages and their merged strings from the file name suffixes. Range over maps yields sorted keys, so merged files ('.toml') take precedence over fragments ('..toml'). */ -}} {{- $langs := slice -}} {{- $langMap := dict -}} {{- range $fileKey, $value := $data -}} {{- $parts := split $fileKey "." -}} {{- $lang := index $parts (sub (len $parts) 1) -}} {{- if not (in $langs $lang) -}} {{- $langs = $langs | append $lang -}} {{- end -}} {{- if reflect.IsMap $value -}} {{- $cur := default (dict) (index $langMap $lang) -}} {{- $langMap = merge $langMap (dict $lang (merge $cur $value)) -}} {{- end -}} {{- end -}} {{- $langs = sort $langs -}} {{- /* Union of all keys across languages — one table row per key */ -}} {{- $allKeys := slice -}} {{- range $lang := $langs -}} {{- $strings := default (dict) (index $langMap $lang) -}} {{- range $key, $entry := $strings -}} {{- if not (in $allKeys $key) -}} {{- $allKeys = $allKeys | append $key -}} {{- end -}} {{- end -}} {{- end -}} {{- $allKeys = sort $allKeys -}}
{{- range $lang := $langs -}} {{- $strings := default (dict) (index $langMap $lang) -}} {{- end -}} {{- range $key := $allKeys -}} {{- range $lang := $langs -}} {{- $strings := default (dict) (index $langMap $lang) -}} {{- $entry := default (dict) (index $strings $key) -}} {{- end -}} {{- end -}}
